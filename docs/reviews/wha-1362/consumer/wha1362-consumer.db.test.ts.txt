let orderNumber = 0;

async function bill() {
  return tenancy.withTenantDb(orgA, async (db) => {
    const order = await db.restaurantOrder.create({
      data: {
        branchId: branchA,
        staffProfileId: staff,
        deviceId: device,
        orderNumber: ++orderNumber,
        businessDate: date,
        channel: 'DINE_IN',
        status: 'SERVED',
      },
    });
    const item = await db.restaurantOrderItem.create({
      data: {
        orderId: order.id,
        productId: product,
        name: 'Meal',
        quantity: 2,
        unitPrice: '12',
        status: 'SERVED',
      },
    });
    return db.restaurantBill.create({
      data: {
        orderId: order.id,
        branchId: branchA,
        openerStaffProfileId: staff,
        deviceId: device,
        businessDate: date,
        status: 'SETTLED',
        receiptNumber: orderNumber,
        settledAt: date,
        subtotal: '24',
        taxTotal: '2',
        serviceChargeTotal: '1',
        deliveryFeeTotal: '3',
        compTotal: '4',
        discountTotal: '0.50',
        roundingAdjustment: '0.05',
        grandTotal: '25.55',
        lines: {
          create: {
            orderItemId: item.id,
            name: 'Meal',
            quantity: 2,
            unitPrice: '12',
            lineSubtotal: '24',
            compAmount: '4',
            taxAmount: '2',
            serviceChargeAmount: '1',
            lineTotal: '20',
          },
        },
        tenders: {
          create: {
            type: 'CASH',
            amount: '25.55',
            tipAmount: '1.50',
            cashReceived: '30',
            changeGiven: '4.45',
            staffProfileId: staff,
            deviceId: device,
            businessDate: date,
          },
        },
      },
    });
  });
}

function args(
  db: TerminalContext['db'],
  billId: string,
  branchId = branchA,
  organizationId = orgA
): MoneyCommandArgs<typeof receiptPrintBody> {
  return {
    ctx: {
      db,
      organizationId,
      device: { id: device, name: 'Counter', kind: 'POS', branchId },
    } as TerminalContext,
    data: { billId, action: 'print' },
    actor: { staffProfileId: staff, deviceId: device, idempotencyKey: `dispatch-${billId}` },
    now: date,
  };
}
