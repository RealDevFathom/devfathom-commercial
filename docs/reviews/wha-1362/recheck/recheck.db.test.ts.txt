// @vitest-environment node
import {randomUUID} from 'node:crypto';
import {beforeAll,afterAll,it,expect} from 'vitest';
import {PrintBridgeService,claimNextPrintJob} from '@/lib/restaurant/printing/service';
import {PrintJobSchema} from '@/lib/restaurant/printing/contract';
import {DEFAULT_RECEIPT_LAYOUT} from '@/lib/receipts/configuration';
import {SAMPLE_RECEIPT,buildReceiptPreview} from '@/lib/receipts/preview';
if (!process.env.TEST_DATABASE_URL) throw new Error('Isolated DB required');
process.env.DATABASE_URL=process.env.TEST_DATABASE_URL;
const org='wha1362-'+randomUUID(), branch='branch-'+org;
let tenancy:typeof import('@/lib/tenancy/scoped-db');
const run=<T>(f:(db:import('@/lib/tenancy/scoped-db').ScopedDb)=>Promise<T>)=>tenancy.withTenantDb(org,f);
const config={branchId:branch,routes:[{documentType:'RECEIPT' as const,stationId:null,printerId:'queue:///Receipt_Printer',enabled:true}]};
const document={type:'RECEIPT' as const,reference:'REVIEW',title:'SYNTHETIC',createdAt:'2026-09-05T12:00:00Z',lines:[{quantity:1,label:'Tea'}]};
beforeAll(async()=>{tenancy=await import('@/lib/tenancy/scoped-db');const {assertRlsEnforcingRole}=await import('@/lib/testing/rls-role-guard');await assertRlsEnforcingRole(tenancy.withPlatformDb);await tenancy.withPlatformDb(db=>db.organization.create({data:{id:org,slug:org,legalName:org,displayName:org}}));await run(db=>db.restaurantBranch.create({data:{id:branch,organizationId:org,name:branch}}));});
afterAll(async()=>{if(tenancy)await tenancy.withPlatformDb(db=>db.organization.deleteMany({where:{id:org}}));});

it.each([58,80] as const)('snapshot retains complete %smm financial preview after caller mutation',async(paperWidth)=>{
 const layout={...DEFAULT_RECEIPT_LAYOUT,paperWidth}; const facts=structuredClone(SAMPLE_RECEIPT);
 const expected=buildReceiptPreview(layout,facts);
 await run(db=>new PrintBridgeService(db).enqueue(org,config,{branchId:branch,stationId:null,idempotencyKey:randomUUID(),document:{...document,receiptSnapshot:{layout,facts}}}));
 layout.footerText='CHANGED';facts.totals=['CHANGED'];facts.payments=['CHANGED'];
 const claim=await run(db=>claimNextPrintJob(db,branch,'review-device'));
 expect(Buffer.from(claim!.bytes,'base64').subarray(8,-7).toString()).toBe(expected);
 await run(db=>db.restaurantPrintJob.deleteMany({}));
});
it('direct enqueue rejects Unicode and malformed text with no persisted row',async()=>{
 for(const [label,code,fallback] of [['چائے','UNSUPPORTED_PRINT_TEXT','BROWSER_OS'],['\x1b@','INVALID_PRINT_DOCUMENT',null]] as const){
 const idempotencyKey=randomUUID();
 await expect(run(db=>new PrintBridgeService(db).enqueue(org,config,{branchId:branch,stationId:null,idempotencyKey,document:{...document,lines:[{quantity:1,label}]}}))).rejects.toMatchObject({code,fallback});
 expect(await run(db=>db.restaurantPrintJob.count({where:{idempotencyKey}}))).toBe(0);
 }
});
it('full invalid batch is held without attempts; next poll reaches valid work; uncertain rows untouched',async()=>{
 const invalid={...document,lines:[{quantity:1,label:'چائے'}]};
 for(let i=0;i<11;i++) await run(db=>db.restaurantPrintJob.create({data:{organizationId:org,branchId:branch,idempotencyKey:randomUUID(),documentType:'RECEIPT',document:i===10?{...document,title:'\x1b@'}:invalid,printerId:config.routes[0].printerId,createdAt:new Date('2026-09-01T00:00:00Z')}}));
 const uncertain=await run(db=>db.restaurantPrintJob.create({data:{organizationId:org,branchId:branch,idempotencyKey:randomUUID(),documentType:'RECEIPT',document:invalid,printerId:config.routes[0].printerId,attemptCount:1,lastAttemptAt:new Date('2026-09-01T00:00:00Z')}}));
 await run(db=>new PrintBridgeService(db).enqueue(org,config,{branchId:branch,stationId:null,idempotencyKey:randomUUID(),document}));
 expect(await run(db=>claimNextPrintJob(db,branch,'review-device'))).toBeNull();
 const claimed=await run(db=>claimNextPrintJob(db,branch,'review-device'));
 expect(claimed).not.toBeNull();expect(Buffer.from(claimed!.bytes,'base64').toString()).toContain('Tea');
 const held=await run(db=>db.restaurantPrintJob.findMany({where:{status:'OFFLINE'}}));
 expect(held).toHaveLength(11);expect(held.every(r=>r.attemptCount===0&&r.lastAttemptAt===null&&r.printedAt===null&&r.byteCount===null)).toBe(true);
 expect(await run(db=>db.restaurantPrintJob.findUniqueOrThrow({where:{id:uncertain.id}}))).toEqual(uncertain);
 expect(await run(db=>claimNextPrintJob(db,branch,'review-device',new Date('2027-01-01')))).toBeNull();
});
